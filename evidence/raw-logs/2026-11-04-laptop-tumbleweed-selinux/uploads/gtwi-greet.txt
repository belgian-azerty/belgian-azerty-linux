== keyd conf
# be-azerty-fix: Windows-style AltGr for Belgian keyboards.
#
# On Windows, Ctrl+Alt is the same as AltGr, so Belgians type @ with
# Ctrl+Alt+2, € with Ctrl+Alt+E, { } with Ctrl+Alt+9 / 0, and so on.
# Linux only honours the right Alt key (AltGr), and the toolkits (GTK, Qt)
# refuse to treat any key press that still carries Ctrl or Alt as text.
#
# keyd works below X11/Wayland (it rewrites the key events at the evdev
# level), so this file makes Ctrl+Alt+<key> arrive at the desktop as
# AltGr+<key> for exactly the 15 keys that carry an AltGr symbol on the
# Windows "Belgian (Period)" layout. Every other Ctrl+Alt combination
# (Ctrl+Alt+T, Ctrl+Alt+Del, Ctrl+Alt+F3, Ctrl+Alt+arrows ...) is passed
# through unchanged. Works in X11, Wayland, the text console, and on the
# login screen, with Caps Lock on or off, with any Ctrl and the left Alt.
#
# Keys are named after the US legend of the physical key (keyd convention):
#   1..0 = top row (& é " ' ( § è ! ç à),  e = E,
#   leftbrace = ^ ¨ (right of P),  rightbrace = $ *,  apostrophe = ù %,
#   backslash = µ £,  slash = = +,  102nd = < >.
#
# Windows differs from the Linux Belgian layout on two keys: AltGr+4 is {
# and AltGr+5 is [ on Windows (Linux gives ¼ ½). Both the AltGr and the
# Ctrl+Alt paths below use the Windows meaning.

[ids]
*

[altgr]
4 = G-9
5 = G-leftbrace

[control+alt]
1 = G-1
2 = G-2
3 = G-3
4 = G-9
5 = G-leftbrace
6 = G-6
9 = G-9
0 = G-0
e = G-e
leftbrace = G-leftbrace
rightbrace = G-rightbrace
apostrophe = G-apostrophe
backslash = G-backslash
slash = G-slash
102nd = G-102nd
